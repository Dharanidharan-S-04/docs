 post method to add the issuer in certify db



create oidckeystore.p12
1. Create the keystore with a key pair
bash
mkdir certs && cd certs
keytool -genkeypair -alias mpartner-default-mimoto-insurance-oidc -keyalg RSA -keysize 2048 -sigalg SHA256withRSA -validity 3650 -storetype PKCS12 -keystore oidckeystore.p12 -storepass mosip123 -keypass mosip123 -dname "CN=mimoto-insurance-oidc, O=YourOrg, C=IN"

ls
oidckeystore.p12

keytool -exportcert -rfc -alias mpartner-default-mimotooidc \
  -keystore oidckeystore.p12 -storepass mosip123 -file oidc-cert.pem

N=$(openssl x509 -in oidc-cert.pem -noout -modulus | cut -d= -f2 \
    | xxd -r -p | base64 -w0 | tr '+/' '-_' | tr -d '=')

cat > oidc-public-jwk.json <<EOF
{
  "e": "AQAB",
  "n": "$N",
  "alg": "RS256",
  "kid": "",
  "kty": "RSA",
  "use": "sig"
}
EOF

keytool -list -keystore oidckeystore.p12 -storepass mosip123
It should show one entry: mpartner-default-mimotooidc, PrivateKeyEntry.

kubectl create secret generic mimotooidc \
  --from-file=oidckeystore.p12=./oidckeystore.p12 \
  -n injiweb --dry-run=client -o yaml | kubectl apply -f -
. Configure Mimoto and restart
•	client_alias in mimoto-issuers-config.json is mpartner-default-mimotooidc
•	client_id is the clientId from step 4
•	oidc_p12_password is mosip123

kubectl rollout restart deployment/mimoto -n injiweb
kubectl exec -n injiweb deploy/mimoto -- keytool -list -keystore certs/oidckeystore.p12 -storepass mosip123
